<script>

const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');

const overlay = document.getElementById('overlay');
const scoreDisplay = document.getElementById('scoreDisplay');

const W = canvas.width;
const H = canvas.height;

let state = 'start';
let frame = 0;
let score = 0;

let best = Number(
  localStorage.getItem('flappyBest') || 0
) || 0;


// Bird
const bird = {
  x: 90,
  y: H / 2,
  radius: 14,
  velocity: 0,
  gravity: 0.5,
  flapStrength: -8.5,
  rotation: 0
};


// Pipes
const pipeWidth = 60;
const pipeGap = 150;
const pipeSpeed = 2.6;

let pipes = [];
let pipeTimer = 0;

const pipeInterval = 95;


// Ground
const groundHeight = 80;
let groundOffset = 0;


function resetGame() {

  bird.y = H / 2;
  bird.velocity = 0;
  bird.rotation = 0;

  pipes = [];
  pipeTimer = 0;

  score = 0;
  frame = 0;

  scoreDisplay.textContent = '0';
}


function flap() {

  if (state === 'start') {

    state = 'playing';

    overlay.style.display = 'none';

    resetGame();

    bird.velocity = bird.flapStrength;

  }

  else if (state === 'playing') {

    bird.velocity = bird.flapStrength;

  }

  else if (state === 'gameover') {

    state = 'start';

    overlay.style.display = 'flex';

    overlay.innerHTML = `
      <h1>🐦 Flappy Bird</h1>
      <p>Press SPACE, click, or tap to flap</p>
      <p>Avoid the pipes and the ground!</p>
      <div class="big">Click to Start</div>
    `;

  }

}


function spawnPipe() {

  const minTop = 60;

  const maxTop =
    H - groundHeight - pipeGap - 60;

  const topHeight =
    Math.random() *
    (maxTop - minTop) +
    minTop;

  pipes.push({

    x: W,

    top: topHeight,

    bottom: topHeight + pipeGap,

    scored: false

  });

}


function update() {

  frame++;

  groundOffset -= pipeSpeed;

  if (groundOffset <= -24) {
    groundOffset = 0;
  }

  if (state !== 'playing') return;


  // Bird physics
  bird.velocity += bird.gravity;

  bird.y += bird.velocity;

  bird.rotation =
    Math.max(
      -0.5,
      Math.min(1.2, bird.velocity / 10)
    );


  // Pipes
  pipeTimer++;

  if (pipeTimer >= pipeInterval) {

    pipeTimer = 0;

    spawnPipe();

  }


  for (let i = pipes.length - 1; i >= 0; i--) {

    const p = pipes[i];

    p.x -= pipeSpeed;


    // Scoring
    if (
      !p.scored &&
      p.x + pipeWidth < bird.x
    ) {

      p.scored = true;

      score++;

      scoreDisplay.textContent = score;

    }


    // Remove offscreen
    if (p.x + pipeWidth < 0) {

      pipes.splice(i, 1);

      continue;

    }


    // Collision
    const birdLeft =
      bird.x - bird.radius;

    const birdRight =
      bird.x + bird.radius;

    const birdTop =
      bird.y - bird.radius;

    const birdBottom =
      bird.y + bird.radius;


    if (
      birdRight > p.x &&
      birdLeft < p.x + pipeWidth
    ) {

      if (
        birdTop < p.top ||
        birdBottom > p.bottom
      ) {

        gameOver();

      }

    }

  }


  // Ground collision
  if (
    bird.y + bird.radius >
    H - groundHeight
  ) {

    bird.y =
      H - groundHeight - bird.radius;

    gameOver();

  }


  // Ceiling collision
  if (
    bird.y - bird.radius < 0
  ) {

    bird.y = bird.radius;

    bird.velocity = 0;

  }

}


function gameOver() {

  if (state !== 'playing') return;

  state = 'gameover';


  if (score > best) {

    best = score;

    localStorage.setItem(
      'flappyBest',
      String(best)
    );

  }


  overlay.style.display = 'flex';

  overlay.innerHTML = `

    <h1>💥 Game Over</h1>

    <p>Score: ${score}</p>

    <p>Best: ${best}</p>

    <div class="big">
      Click to Play Again
    </div>

  `;

}


function drawBackgroundClouds() {

  ctx.fillStyle =
    'rgba(255,255,255,0.6)';

  for (let i = 0; i < 3; i++) {

    const cx =
      ((frame * 0.3) + i * 160)
      % (W + 80) - 40;

    const cy =
      80 + i * 60;


    ctx.beginPath();

    ctx.ellipse(
      cx,
      cy,
      30,
      14,
      0,
      0,
      Math.PI * 2
    );

    ctx.ellipse(
      cx + 20,
      cy + 4,
      20,
      10,
      0,
      0,
      Math.PI * 2
    );

    ctx.fill();

  }

}


function drawPipes() {

  pipes.forEach(p => {

    const grad =
      ctx.createLinearGradient(
        p.x,
        0,
        p.x + pipeWidth,
        0
      );

    grad.addColorStop(
      0,
      '#4caf50'
    );

    grad.addColorStop(
      0.5,
      '#7ed957'
    );

    grad.addColorStop(
      1,
      '#388e3c'
    );

    ctx.fillStyle = grad;


    // Top pipe
    ctx.fillRect(
      p.x,
      0,
      pipeWidth,
      p.top
    );

    ctx.fillRect(
      p.x - 4,
      p.top - 20,
      pipeWidth + 8,
      20
    );


    // Bottom pipe
    ctx.fillRect(
      p.x,
      p.bottom,
      pipeWidth,
      H - groundHeight - p.bottom
    );

    ctx.fillRect(
      p.x - 4,
      p.bottom,
      pipeWidth + 8,
      20
    );


    ctx.strokeStyle = '#2e7d32';

    ctx.lineWidth = 2;

    ctx.strokeRect(
      p.x,
      0,
      pipeWidth,
      p.top
    );

    ctx.strokeRect(
      p.x,
      p.bottom,
      pipeWidth,
      H - groundHeight - p.bottom
    );

  });

}


function drawGround() {

  ctx.fillStyle = '#ded895';

  ctx.fillRect(
    0,
    H - groundHeight,
    W,
    groundHeight
  );


  ctx.fillStyle = '#8bc34a';

  ctx.fillRect(
    0,
    H - groundHeight,
    W,
    10
  );


  ctx.fillStyle = '#c9c07a';

  for (
    let x = groundOffset;
    x < W;
    x += 24
  ) {

    ctx.fillRect(
      x,
      H - groundHeight + 14,
      14,
      6
    );

  }

}


function drawBird() {

  ctx.save();

  ctx.translate(
    bird.x,
    bird.y
  );

  ctx.rotate(
    bird.rotation
  );


  // Body
  ctx.fillStyle = '#ffcc00';

  ctx.beginPath();

  ctx.ellipse(
    0,
    0,
    bird.radius,
    bird.radius - 1,
    0,
    0,
    Math.PI * 2
  );

  ctx.fill();


  ctx.strokeStyle = '#e0a800';

  ctx.lineWidth = 2;

  ctx.stroke();


  // Wing
  const wingFlap =
    Math.sin(frame * 0.4) * 4;

  ctx.fillStyle = '#ffe066';

  ctx.beginPath();

  ctx.ellipse(
    -4,
    2 + wingFlap,
    8,
    5,
    0.3,
    0,
    Math.PI * 2
  );

  ctx.fill();


  // Eye
  ctx.fillStyle = 'white';

  ctx.beginPath();

  ctx.arc(
    6,
    -4,
    4,
    0,
    Math.PI * 2
  );

  ctx.fill();
